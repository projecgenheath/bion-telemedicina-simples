   225→const MESES = ["Jan", "Fev", "Mar", "Abr", "Mai", "Jun", "Jul", "Ago", "Set", "Out", "Nov", "Dez"];
   226→
   227→const mesmoDia = (a: Date, b: Date) =>
   228→  a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate();
   229→
   230→const fmtCurta = (iso: string) => {
   231→  const d = new Date(iso);
   232→  const hoje = new Date();
   233→  const amanha = new Date();
   234→  amanha.setDate(hoje.getDate() + 1);
   235→  if (mesmoDia(d, hoje)) return "Hoje";
   236→  if (mesmoDia(d, amanha)) return "Amanhã";
   237→  return `${d.getDate()} ${MESES[d.getMonth()]}`;
   238→};
   239→
   240→const fmtHora = (iso: string) =>
   241→  new Date(iso).toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" });
   242→
   243→const fmtLonga = (iso: string) => {
   244→  const d = new Date(iso);
   245→  return `${String(d.getDate()).padStart(2, "0")} ${MESES[d.getMonth()]} ${d.getFullYear()}`;
   246→};
   247→
   248→const fmtQuando = (iso: string) => {
   249→  const d = new Date(iso);
   250→  return `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}/${d.getFullYear()} ${fmtHora(iso)}`;
   251→};
   252→
   253→const fmtTicketData = (iso: string) => {
   254→  const d = new Date(iso);
   255→  const hoje = new Date();
   256→  const ontem = new Date();
   257→  ontem.setDate(hoje.getDate() - 1);
   258→  if (mesmoDia(d, hoje)) return `Hoje às ${fmtHora(iso)}`;
   259→  if (mesmoDia(d, ontem)) return `Ontem às ${fmtHora(iso)}`;
   260→  return `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}/${d.getFullYear()} às ${fmtHora(iso)}`;
   261→};
   262→
   263→const fmtDataBR = (iso: string) => {
   264→  const d = new Date(iso);
   265→  return `${String(d.getDate()).padStart(2, "0")}/${String(d.getMonth() + 1).padStart(2, "0")}/${d.getFullYear()}`;
   266→};
   267→
   268→const fmtValorBRL = (v: number) =>
   269→  Number.isInteger(v) ? `R$ ${v}` : `R$ ${v.toFixed(2).replace(".", ",")}`;

   366→async function api<T>(url: string, init?: RequestInit): Promise<T | null> {
   367→  try {
   368→    const res = await fetch(url, {
   369→      headers: { "Content-Type": "application/json" },
   370→      ...init,
   371→    });
   372→    const json = res.status === 204 ? null : await res.json().catch(() => null);
   373→    if (!res.ok) {
   374→      const msg = (json as { erro?: string } | null)?.erro ?? "Não foi possível concluir a operação.";
   375→      toast.error(msg);
   376→      return null;
   377→    }
   378→    return json as T;
   379→  } catch {
   380→    toast.error("Falha de conexão com o servidor.");
   381→    return null;
   382→  }
   383→}
